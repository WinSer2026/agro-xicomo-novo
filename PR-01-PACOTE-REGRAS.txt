--------------------------------------------------------------------------------
1. O QUE ESTE PR FAZ, EM LINGUAGEM DE BALCAO
--------------------------------------------------------------------------------

Hoje, qualquer pessoa que abra o site e saiba o endereco do projecto consegue
pedir ao Firebase: "apaga esta venda". O Firebase responde que sim, porque as
regras so perguntam "esta autenticado?" — e a aplicacao cria uma sessao anonima
para qualquer visitante.

Este PR muda isso em 8 colecções (mais a auditoria, que passa a ter delete
condicionado). Em linguagem de balcao:

  - O LIVRO DE VENDAS deixa de poder ser arrancado. Pode escrever-se uma venda
    nova, pode anular-se (que e uma escrita normal, com estorno), mas nao ha
    forma de apagar a pagina.
  - O LIVRO DE NUMERACAO (counters) fica trancado. Se alguem apagasse um
    contador, o sistema voltaria a emitir numeros ja usados — duas facturas com
    o mesmo numero.
  - As DEFINICOES (nome da empresa, NUIT, codigo de gestor) deixam de poder ser
    apagadas por um clique anonimo.
  - O ARQUIVO DE AUDITORIA so pode ser "movido" — nunca apagado directamente.
    (E aqui esta o unico ponto delicado: ver seccao 6.)

O QUE ESTE PR NAO FAZ (e importante nao vender ilusoes):
  - Nao impede LER. Uma sessao anonima continua a poder ler vendas, salarios e
    a colecção de credenciais. Isso so fecha na Onda 3 (PR-11/PR-12), porque
    exige mudar o login.
  - Nao impede ALTERAR o total de uma venda ja emitida. Isso e um PR seguinte
    (PR-01c), com regras por campo.
  - Nao resolve as palavras-passe no codigo (isso e o PR-03).

Mesmo assim, e o PR com melhor relacao risco/beneficio de todo o plano: em
duas horas, o pior cenario ("apagaram-me o historico") deixa de estar a um
clique de distancia.

rules_version = '2';
// ---------------------------------------------------------------------------
// Regras do Firestore — AGRO-XICOMO
//
// ATENCAO: exigem autenticacao (Firebase Anonymous Auth ligado no projeto).
// A versao anterior era `allow read, write: if true`, o que expunha TODA a
// base — vendas, credenciais (mesmo em hash), salarios — a qualquer pessoa que
// tivesse o projectId (que esta no bundle do frontend).
//
// Ultima alteracao: 2026-11-09 — PR-01.
//   * sales, counters, settings, transformations, customers, attendance,
//     losses e payroll_payments deixam de aceitar delete do cliente.
//   * audit_logs so aceita delete quando o documento ja foi copiado para
//     audit_logs_archive na MESMA operacao (getAfter).
//   * credentials e products mantem-se: o login le os hashes e a eliminacao
//     de produto e uma funcao da aplicacao. Fecham em PR-01b e na Onda 3.
//
// Antes de publicar (`firebase deploy --only firestore:rules`):
//   1. Console Firebase -> Authentication -> Sign-in method -> ativar "Anonimo".
//   2. Abrir a aplicacao e confirmar que os dados aparecem.
//   3. Testar no simulador da consola (Rules Playground) os 6 casos da seccao 5.
//   4. So entao publicar. Ver docs/seguranca.md.
//
// O ficheiro com as regras antigas esta em firestore.rules.legacy-open, apenas
// para consulta/historico. NAO o publique.
// ---------------------------------------------------------------------------
service cloud.firestore {
  match /databases/{database}/documents {

    // Negado por omissao a tudo o que nao esteja listado abaixo.
    match /{document=**} {
      allow read, write: if false;
    }

    // --- Operacao normal ----------------------------------------------------

    match /products/{docId} {
      allow read, create, update: if request.auth != null;
      // MANTER por agora: StockModule.tsx:246 apaga produtos.
      // PR-01b troca por soft-delete (deletedAt) e este delete passa a false.
      allow delete: if request.auth != null;
    }

    match /sales/{docId} {
      allow read, create, update: if request.auth != null;
      allow delete: if false;                                   // <== MUDOU
    }

    match /losses/{docId} {
      allow read, create, update: if request.auth != null;
      allow delete: if false;                                   // <== MUDOU
    }

    match /transformations/{docId} {
      allow read, create, update: if request.auth != null;
      // dbService.deleteTransformation existe (firebaseDB.ts:2639) mas nao e
      // chamado por nenhum ecra. Cancelar e um update (status), nao um delete.
      allow delete: if false;                                   // <== MUDOU
    }

    match /employees/{docId} {
      allow read, create, update: if request.auth != null;
      // MANTER por agora: HRModule.tsx:428 elimina funcionarios.
      // PR-01b troca por desactivacao (deletedAt) para preservar historico
      // salarial e disciplinar.
      allow delete: if request.auth != null;
    }

    match /attendance/{docId} {
      allow read, create, update: if request.auth != null;
      allow delete: if false;                                   // <== MUDOU
    }

    match /customers/{docId} {
      allow read, create, update: if request.auth != null;
      allow delete: if false;                                   // <== MUDOU
    }

    match /paused_sales/{docId} {
      allow read, write: if request.auth != null;
    }

    match /active_cart/{docId} {
      allow read, write: if request.auth != null;
    }

    match /settings/{docId} {
      allow read, create, update: if request.auth != null;
      // Contem nome da empresa, NUIT e o codigo de gestor de stock.
      allow delete: if false;                                   // <== MUDOU
    }

    match /counters/{docId} {
      allow read, create, update: if request.auth != null;
      // Numeracao fiscal. Se um contador for apagado, o sistema reemite
      // numeros ja usados — duas facturas com o mesmo numero.
      allow delete: if false;                                   // <== MUDOU
    }

    match /deletions/{docId} {
      allow read, write: if request.auth != null;
    }

    match /payroll_payments/{docId} {
      allow read, create, update: if request.auth != null;
      allow delete: if false;                                   // <== MUDOU
    }

    match /active_sessions/{docId} {
      allow read, write: if request.auth != null;
    }

    // --- Credenciais da aplicacao -------------------------------------------
    // O login e feito na aplicacao (nao no Firebase Auth): o cliente le o hash
    // para o comparar. Enquanto isso for verdade, NAO se pode negar a leitura
    // aqui — partiria o login de toda a equipa.
    // Fecho definitivo: PR-11/PR-12 (login por Cloud Function + custom claims).
    match /credentials/{docId} {
      allow read, create, update, delete: if request.auth != null;
    }

    // --- Auditoria -----------------------------------------------------------
    // Os registos nao podem ser EDITADOS. Podem ser ARQUIVADOS: o arquivo copia
    // o documento para `audit_logs_archive` e apaga o original, na mesma
    // operacao (firebaseDB.ts:3115-3160).
    // Para impedir que se apague um log "a mao", o delete passa a exigir prova
    // de que a copia existe DEPOIS da operacao (getAfter).
    match /audit_logs/{docId} {
      allow read, create: if request.auth != null;
      allow update: if false;
      allow delete: if request.auth != null
        && getAfter(/databases/$(database)/documents/audit_logs_archive/$(docId)).data.archivedFrom == 'audit_logs';  // <== MUDOU
    }

    match /audit_logs_archive/{docId} {
      allow read, create: if request.auth != null;
      allow update, delete: if false;
    }

    // --- Livro de movimentos de stock ---------------------------------------
    match /stock_movements/{docId} {
      allow read, create: if request.auth != null;
      allow update, delete: if false;
    }

    // --- Dividas de clientes ------------------------------------------------
    match /debts/{docId} {
      allow read, create, update: if request.auth != null;
      allow delete: if false;
    }

    // --- Livro de caixa -----------------------------------------------------
    match /cash_movements/{docId} {
      allow read, create: if request.auth != null;
      allow update, delete: if false;
    }
  }
}

6. O PONTO DELICADO: o arquivo de logs (getAfter)
--------------------------------------------------------------------------------
Porque e delicado
  O arquivo de auditoria, escrito ha meses, faz uma unica operacao em lote:
  copia o log para audit_logs_archive e apaga o original. Se eu negar o delete
  puro e simples, o arquivo deixa de funcionar — e o botao "Arquivar registos"
  do modulo tecnico passa a dar erro.

Como resolvo
  `getAfter(...)` pergunta ao Firebase: "depois desta operacao, o documento de
  arquivo vai existir e ter archivedFrom = 'audit_logs'?" Se sim, o delete e
  autorizado. Se alguem tentar apagar um log sozinho, nao ha copia prevista e a
  regra nega.

Riscos reais (nao escondo nenhum)
  R1. Se a versao do Firestore nao suportar getAfter neste contexto, a regra
      da erro -> delete negado -> o arquivo parte.
      Probabilidade: baixa (getAfter existe nas regras v2). O teste 5 do
      simulador detecta isto ANTES de publicar.
  R2. Se o arquivo mudar de forma (por exemplo, passar a apagar sem copiar),
      volta a falhar. Por isso o teste automatizado (seccao 7) passa a vigiar a
      presenca de `getAfter` e de `archivedFrom`.
  R3. Um erro aqui afecta apenas o arquivo de logs, nao vendas nem stock.

PLANO B (se o simulador falhar ou se algo correr mal)
  Trocar as 3 linhas do audit_logs por:
      allow delete: if request.auth != null;
  e manter as outras 8 mudancas (que nao dependem de nada disto). Registar como
  risco aceite ate o arquivo passar para uma funcao de servidor (Onda 3).

// --- PR-01: parser de blocos por chaves balanceadas (nao por regex guloso) ---
// NOTA: a primeira versao deste teste usava um regex que, com as regras escritas
// numa so linha por colecção, engolia as colecções vizinhas e podia dar "verde"
// falso. Este parser foi validado contra as regras antigas e as novas.
function blocosDeRegras(texto: string): Map<string, string> {
  const mapa = new Map<string, string>();
  const re = /match\s+\/([a-z_]+)\/\{([a-zA-Z]+)\}\s*\{/g;
  let m: RegExpExecArray | null;
  while ((m = re.exec(texto)) !== null) {
    let i = re.lastIndex;
    let nivel = 1;
    const inicio = i;
    while (i < texto.length && nivel > 0) {
      if (texto[i] === '{') nivel++;
      else if (texto[i] === '}') nivel--;
      i++;
    }
    mapa.set(m[1], texto.slice(inicio, i - 1));
  }
  return mapa;
}
